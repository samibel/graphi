```go
// VisitParents visits all parents of the command and invokes fn on each parent.
func (c *Command) VisitParents(fn func(*Command)) {
	if c.HasParent() {
		fn(c.Parent())
		c.Parent().VisitParents(fn)
	}
}
```

3. Recurse upward: it then calls c.Parent().VisitParents(fn), moving one level closer to the root and repeating the same steps.
